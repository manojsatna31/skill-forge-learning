Setting Up Your Environment

Prerequisite Setups

1️⃣ Install Java (JDK 17 or later)

Java is the foundation. We use the latest LTS version (JDK 17 or 21).

🖥️ Windows
  1. Download from Adoptium.
  2. Run the installer and set JAVA_HOME.
  3. Add bin to PATH.
java -version
        javac -version
🍎 macOS / 🐧 Linux
  1. Use brew install openjdk@17 (macOS) or sudo apt install openjdk-17-jdk (Ubuntu).
  2. Set JAVA_HOME in ~/.bashrc or ~/.zshrc.
java -version
        javac -version

2️⃣ Install IntelliJ IDEA (Community Edition)

IntelliJ is the most popular Java IDE, with excellent refactoring and debugging tools.

  • Download from jetbrains.com.
  • Install and launch.
  • Create a new project and select JDK 17.
  • Install plugins: Lombok, Spring Boot (optional).

3️⃣ Install MySQL (or use Docker)

MySQL is used for database projects.

  • Windows: Download MySQL Installer and install server + Workbench.
  • macOS: brew install mysql.
  • Linux: sudo apt install mysql-server.
  • Start the service and verify: mysql -u root -p.

4️⃣ Install Maven (Build Tool)

Maven manages dependencies and builds your projects.

  • Download from maven.apache.org.
  • Unzip and set MAVEN_HOME.
  • Add bin to PATH.
mvn --version

5️⃣ Git – Version Control

Same as Python – install Git and configure your name/email.

git config --global user.name "Your Name"
        git config --global user.email "you@example.com"

        }
        balance -= amount;
    }
}

Day 55 – Fork/Join Framework

Concept: Divide‑and‑conquer parallelism. Use ForkJoinPool with RecursiveTask or RecursiveAction.

import java.util.concurrent.*;

// ─── RecursiveTask – returns a result ─────────────────────
class SumTask extends RecursiveTask {
    private static final int THRESHOLD = 10;
    private int[] array;
    private int start;
    private int end;

    SumTask(int[] array, int start, int end) {
        this.array = array;
        this.start = start;
        this.end = end;
    }

    @Override
    protected Integer compute() {
        if (end - start <= THRESHOLD) {
            // Sequential sum
            int sum = 0;
            for (int i = start; i < end; i++) {
                sum += array[i];
            }
            return sum;
        } else {
            // Divide and conquer
            int mid = (start + end) / 2;
            SumTask left = new SumTask(array, start, mid);
            SumTask right = new SumTask(array, mid, end);
            left.fork(); // execute asynchronously
            int rightResult = right.compute();
            int leftResult = left.join();
            return leftResult + rightResult;
        }
    }
}

public class ForkJoinDemo {
    public static void main(String[] args) {
        int[] array = new int[1000];
        for (int i = 0; i < array.length; i++) {
            array[i] = i + 1;
        }

        ForkJoinPool pool = ForkJoinPool.commonPool();
        SumTask task = new SumTask(array, 0, array.length);
        int result = pool.invoke(task);
        System.out.println("Sum: " + result);

        // Using parallel stream (built‑on Fork/Join)
        int parallelSum = java.util.Arrays.stream(array).parallel().sum();
        System.out.println("Parallel sum: " + parallelSum);
    }
}

Day 58 – JDBC Transactions

Concept: Group multiple operations into a single unit of work. Commit only if all succeed, otherwise rollback.

import java.sql.*;

public class TransactionDemo {
    public static void main(String[] args) {
        String url = "jdbc:mysql://localhost:3306/mydb";
        String user = "root";
        String password = "password";

        try (Connection conn = DriverManager.getConnection(url, user, password)) {
            conn.setAutoCommit(false); // Start transaction

            try (PreparedStatement pstmt1 = conn.prepareStatement(
                     "UPDATE accounts SET balance = balance - ? WHERE id = ?");
                 PreparedStatement pstmt2 = conn.prepareStatement(
                     "UPDATE accounts SET balance = balance + ? WHERE id = ?")) {

                // Transfer $100 from account 1 to account 2
                pstmt1.setDouble(1, 100);
                pstmt1.setInt(2, 1);
                int rows1 = pstmt1.executeUpdate();

                pstmt2.setDouble(1, 100);
                pstmt2.setInt(2, 2);
                int rows2 = pstmt2.executeUpdate();

                if (rows1 > 0 && rows2 > 0) {
                    conn.commit();
                    System.out.println("Transaction successful");
                } else {
                    conn.rollback();
                    System.out.println("Transaction rolled back");
                }

            } catch (SQLException e) {
                conn.rollback();
                System.out.println("Error, transaction rolled back");
                e.printStackTrace();
            } finally {
                conn.setAutoCommit(true);
            }

        } catch (SQLException e) {
            e.printStackTrace();
        }
    }
}

Day 77 – Singleton Pattern

Concept: Ensures a class has only one instance and provides a global access point to it.

// ─── Eager Initialization ──────────────────────────────────
class EagerSingleton {
    private static final EagerSingleton INSTANCE = new EagerSingleton();

    private EagerSingleton() {}

    public static EagerSingleton getInstance() {
        return INSTANCE;
    }
}

// ─── Lazy Initialization (Thread‑Safe) ────────────────────
class LazySingleton {
    private static volatile LazySingleton instance;

    private LazySingleton() {}

    public static LazySingleton getInstance() {
        if (instance == null) {
            synchronized (LazySingleton.class) {
                if (instance == null) {
                    instance = new LazySingleton();
                }
            }
        }
        return instance;
    }
}

// ─── Enum Singleton (Recommended) ─────────────────────────
enum EnumSingleton {
    INSTANCE;

    public void doSomething() {
        System.out.println("Doing something...");
    }
}

        EagerSingleton s1 = EagerSingleton.getInstance();
        EagerSingleton s2 = EagerSingleton.getInstance();

Day 79 – Observer Pattern

Concept: Defines a one‑to‑many dependency where a subject notifies multiple observers of state changes.

import java.util.*;

// ─── Subject ────────────────────────────────────────────────
class Stock {
    private String symbol;
    private double price;
    private List observers = new ArrayList<>();

    public Stock(String symbol, double price) {
        this.symbol = symbol;
        this.price = price;
    }

    public void attach(Observer observer) {
        observers.add(observer);
    }

    public void detach(Observer observer) {
        observers.remove(observer);
    }

    public void setPrice(double price) {
        this.price = price;
        notifyAllObservers();
    }

    private void notifyAllObservers() {
        for (Observer observer : observers) {
            observer.update(symbol, price);
        }
    }
}

// ─── Observer Interface ────────────────────────────────────
interface Observer {
    void update(String symbol, double price);
}

// ─── Concrete Observers ────────────────────────────────────
class Investor implements Observer {
    private String name;

    public Investor(String name) {
        this.name = name;
    }

    @Override
    public void update(String symbol, double price) {
        System.out.println(name + " - " + symbol + " price changed to $" + price);
    }
}

// ─── Usage ──────────────────────────────────────────────────
public class ObserverDemo {
    public static void main(String[] args) {

        Investor alice = new Investor("Alice");

        apple.attach(bob);

Day 80 – Strategy Pattern

Concept: Defines a family of algorithms, encapsulates each, and makes them interchangeable at runtime.

// ─── Strategy Interface ────────────────────────────────────
interface SortStrategy {
    void sort(int[] numbers);
}

// ─── Concrete Strategies ──────────────────────────────────
class BubbleSort implements SortStrategy {
    @Override
    public void sort(int[] numbers) {
        for (int i = 0; i < numbers.length - 1; i++) {
            for (int j = 0; j < numbers.length - i - 1; j++) {
                if (numbers[j] > numbers[j + 1]) {
                    int temp = numbers[j];
                    numbers[j] = numbers[j + 1];
                    numbers[j + 1] = temp;
                }
            }
        }
        System.out.println("Bubble sort completed");
    }
}

class QuickSort implements SortStrategy {
    @Override
    public void sort(int[] numbers) {
        quickSort(numbers, 0, numbers.length - 1);
        System.out.println("Quick sort completed");
    }

    private void quickSort(int[] arr, int low, int high) {
        if (low < high) {
            int pi = partition(arr, low, high);
            quickSort(arr, low, pi - 1);
            quickSort(arr, pi + 1, high);
        }
    }

    private int partition(int[] arr, int low, int high) {
        int pivot = arr[high];
        int i = low - 1;
        for (int j = low; j < high; j++) {
            if (arr[j] < pivot) {
                i++;
                int temp = arr[i];
                arr[i] = arr[j];
                arr[j] = temp;
            }
        }
        int temp = arr[i + 1];
        arr[i + 1] = arr[high];
        arr[high] = temp;
        return i + 1;
    }
}

// ─── Context ────────────────────────────────────────────────
class Sorter {
    private SortStrategy strategy;

    public Sorter(SortStrategy strategy) {
        this.strategy = strategy;
    }

    public void setStrategy(SortStrategy strategy) {
        this.strategy = strategy;
    }

    public void sort(int[] numbers) {
        strategy.sort(numbers);
    }
}

// ─── Usage ──────────────────────────────────────────────────
public class StrategyDemo {
    public static void main(String[] args) {
        int[] data = {5, 2, 8, 1, 9, 3};

        Sorter sorter = new Sorter(new BubbleSort());
        sorter.sort(data.clone()); // using clone to keep original

        sorter.setStrategy(new QuickSort());
        sorter.sort(data.clone());
    }
}

Day 96 – Security (OAuth2 / JWT)

Concept: OAuth2 is an authorisation framework; JWT is a token format. Spring Security provides OAuth2 support.

// ─── JWT Utility ───────────────────────────────────────────
import io.jsonwebtoken.*;
import java.util.Date;

public class JwtUtil {
    private static final String SECRET = "mySecretKey";
    private static final long EXPIRATION = 1000 * 60 * 60; // 1 hour

    public static String generateToken(String username) {
        return Jwts.builder()
            .setSubject(username)
            .setIssuedAt(new Date())
            .setExpiration(new Date(System.currentTimeMillis() + EXPIRATION))
            .signWith(SignatureAlgorithm.HS256, SECRET)
            .compact();
    }

    public static String validateToken(String token) {
        try {
            Claims claims = Jwts.parser()
                .setSigningKey(SECRET)
                .parseClaimsJws(token)
                .getBody();
            return claims.getSubject();
        } catch (Exception e) {
            return null;
        }
    }
}

                .anyRequest().authenticated()
            )
            .oauth2ResourceServer(oauth2 -> oauth2.jwt());

    }

import org.apache.spark.sql.Row;
import org.apache.spark.sql.SparkSession;
import static org.apache.spark.sql.functions.*;

public class SparkETLDemo {
    public static void main(String[] args) {
        // ─── Create Spark Session ──────────────────────────────
        SparkSession spark = SparkSession.builder()
            .appName("Java ETL Demo")
            .master("local[*]") // local mode
            .getOrCreate();

        // ─── Read CSV ────────────────────────────────────────────
        Dataset df = spark.read()
            .option("header", "true")
            .option("inferSchema", "true")
            .csv("data/orders.csv");

        df.printSchema();
        df.show(5);

        // ─── Transformations ────────────────────────────────────
        // Filter orders with amount > 100
        Dataset filtered = df.filter(col("amount").gt(100));

        // Aggregate by product
        Dataset aggregated = df.groupBy("product")
            .agg(
                sum("amount").alias("total_sales"),
                count("*").alias("order_count"),
                avg("amount").alias("avg_order")
            )
            .orderBy(col("total_sales").desc());

        aggregated.show();

        // ─── Write to Parquet ────────────────────────────────────
        aggregated.write()
            .mode("overwrite")
            .parquet("output/sales_aggregation.parquet");

        // ─── Spark SQL ───────────────────────────────────────────
        df.createOrReplaceTempView("orders");
        Dataset sqlResult = spark.sql(
            "SELECT product, SUM(amount) as total " +
            "FROM orders " +
            "WHERE amount > 100 " +
            "GROUP BY product " +
            "ORDER BY total DESC"
        );
        sqlResult.show();

        spark.stop();
    }
}

        this.model = model;
        this.registry = registry;
        this.predictionTimer = Timer.builder("prediction.duration")
            .description("Time to make a prediction")
            .register(registry);

            .description("Total predictions")
            .register(registry);

        long start = System.currentTimeMillis();

            predictionTimer.record(duration, java.util.concurrent.TimeUnit.MILLISECONDS);

            return new PredictionResponse(result);
        } catch (Exception e) {
            registry.counter("prediction.errors").increment();
            throw new RuntimeException("Prediction failed: " + e.getMessage());
        }
    }

⚠️ Common Gotchas & Pitfalls

  • Comparing Strings with == – use equals().
  • Floating‑point precision – use BigDecimal for monetary values.
  • Array index out of bounds – check length before access.
  • NullPointerException – always check for null or use Optional.
  • Concurrent modification – don't modify a collection while iterating; use Iterator.remove() or concurrent collections.
  • Memory leak – unused references in caches/listeners – use WeakHashMap or clear references.
  • Overriding equals() without overriding hashCode() – breaks hash‑based collections.
  • Using raw types – always use generics for type safety.
  • Ignoring checked exceptions – catch or declare them.
